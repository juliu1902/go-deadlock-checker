// Paar 1: zwei sends in unterschiedlicher Reihenfolge -> equiv FALSE!
func foo (c1 chan int, c2 chan int){
    c1 <- 1
    c2 <- 2
    close c1
    close c2
}
func bar (c1 chan int, c2 chan int){
    c2 <- 2
    c1 <- 1
    close c2
    close c1
}

var a chan int
var b chan int
foo (a, b)
bar (a, b)

// Paar 2: durch cond-eta -> equiv TRUE!
func foo (c chan int){
    if * then { c <- 1 } else { c <- 2 }
    c <- 3
    close c
}
func bar (c chan int){
    if * then { 
        c <- 1
        c <- 3 
    } else { 
        c <- 2
        c <- 3 
    }
    close c
}

var c chan int
foo (c)
bar (c)
---
// Paar 3: Auch hier dasselbe, in anderer Konstellation -> equiv TRUE!
func foo (c chan int, b bool){
    c <- 0
    if b then { c <- 1 } else { c <- 2 }
    close c
}
func bar (c chan int, b bool){
    if b then { 
        c <- 0
        c <- 1 
    } else { 
        c <- 0 
        c <- 2 
    }
    close c
}

var c chan int
var b bool
foo (c, b)
bar (c, b)

// Paar 4, aliasing test, c wird einmal c1 und einmal c2 zugewiesen, daher c1!;c2! -> dual TRUE!
