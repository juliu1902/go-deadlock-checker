// Paar 1: zwei sends in unterschiedlicher Reihenfolge -> equiv FALSE!
func foo (c1 chan int, c2 chan int){
    c1 <- 1
    c2 <- 2
    close (c1)
    close (c2)
}
func bar (c1 chan int, c2 chan int){
    c2 <- 2
    c1 <- 1
    close (c2)
    close (c1)
}

var b chan int
foo (a, b)

// Paar 2: durch cond-eta -> equiv TRUE!
func foo (c chan int, b bool){
    if b then { c <- 1 } else { c <- 2 }
    c <- 3
    close (c)
}
func bar (c chan int, b bool){
    if b then { 
        c <- 1
        c <- 3 
    } else { 
        c <- 2
        c <- 3 
    }
    close (c)
}

var c chan int
var b bool
